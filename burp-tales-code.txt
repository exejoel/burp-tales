export async function createRestaurantVisit(formData: FormData) {
  const restaurantName = formData.get("restaurantName") as string;
  const cuisine = formData.get("cuisine") as string;
  const city = formData.get("city") as string;
  const date = formData.get("date") as string;
  const rating = formData.get("rating") as string;
  const notes = formData.get("notes") as string;

  if (!restaurantName || !date) {
    throw new Error("Restaurant name and date are required");
  }

  // Find an existing restaurant with this name, or create a new one
  let restaurant = await prisma.restaurant.findFirst({
    where: { name: restaurantName },
  });

  if (!restaurant) {
    restaurant = await prisma.restaurant.create({
      data: { name: restaurantName, cuisine, city },
    });
  }

  await prisma.visit.create({
    data: {
      userId: PLACEHOLDER_USER_ID,
      restaurantId: restaurant.id,
      date: new Date(date),
      rating: rating ? parseInt(rating) : null,
      notes,
    },
  });

  redirect("/restaurants");
}

"use server";

import { prisma } from "@/lib/prisma";
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";
import { scrapeRecipeFromUrl } from "@/lib/scrapeRecipe";
import { requireOwner } from "@/lib/owner";

// The owner's row in the User table. Every recipe belongs to it, since
// only the owner can write (see requireOwner below).
const PLACEHOLDER_USER_ID = "cmtxs8n2l0000muiy9ezqjki9";

// What a create or update action sends back to the form when it can't
// save. On success the action redirects instead, so there's no success state.
export type RecipeFormState = {
  fieldErrors?: {
    title?: string;
    ingredients?: string;
    steps?: string;
  };
  formError?: string;
  // What the user submitted, so the form can show it again after an error.
  values?: {
    title: string;
    steps: string;
  };
};

type RecipeFields = {
  title: string;
  ingredients: string;
  steps: string;
};

// Owner-only too: left open, anyone could make your server fetch any URL
// they choose (server-side request forgery).
export async function fetchRecipeFromUrl(url: string) {
  await requireOwner();
  return scrapeRecipeFromUrl(url);
}

// Not exported, so it's a plain helper, not a Server Action the
// browser could call. Shared by create and update.
function readRecipeFields(
  formData: FormData
):
  | { ok: true; fields: RecipeFields }
  | { ok: false; state: RecipeFormState } {
  const rawTitle = String(formData.get("title") ?? "");
  const rawSteps = String(formData.get("steps") ?? "");

  const title = rawTitle.trim();
  const ingredients = String(formData.get("ingredients") ?? "").trim();
  const steps = rawSteps.trim();

  const fieldErrors: NonNullable<RecipeFormState["fieldErrors"]> = {};
  if (!title) fieldErrors.title = "Add a title.";
  if (!ingredients) fieldErrors.ingredients = "Add at least one ingredient.";
  if (!steps) fieldErrors.steps = "Add the steps.";

  if (Object.keys(fieldErrors).length > 0) {
    return {
      ok: false,
      state: { fieldErrors, values: { title: rawTitle, steps: rawSteps } },
    };
  }
  return { ok: true, fields: { title, ingredients, steps } };
}

// With useActionState, React calls the action as (previousState, formData).
export async function createRecipe(
  _prevState: RecipeFormState,
  formData: FormData
): Promise<RecipeFormState> {
  // Check first, before reading or validating anything.
  await requireOwner();

  const result = readRecipeFields(formData);
  if (!result.ok) return result.state;

  const { fields } = result;
  const sourceUrl = String(formData.get("sourceUrl") ?? "");
  const imageUrl = String(formData.get("imageUrl") ?? "");

  try {
    await prisma.recipe.create({
      data: {
        ...fields,
        sourceUrl: sourceUrl || null,
        imageUrl: imageUrl || null,
        userId: PLACEHOLDER_USER_ID,
      },
    });
  } catch (err) {
    console.error("createRecipe failed:", err);
    return {
      formError: "Couldn't save the recipe. Try again.",
      values: { title: fields.title, steps: fields.steps },
    };
  }

  // Outside the try/catch on purpose: redirect() works by throwing a
  // special error, and a catch block would swallow it.
  revalidatePath("/recipes");
  redirect("/recipes");
}

// The edit page binds the id: updateRecipe.bind(null, recipe.id).
// Bound values travel through the browser, so treat `id` like any user
// input: matching on userId too means you can only edit your own recipes.
export async function updateRecipe(
  id: string,
  _prevState: RecipeFormState,
  formData: FormData
): Promise<RecipeFormState> {
  // Check first, before reading or validating anything.
  await requireOwner();

  const result = readRecipeFields(formData);
  if (!result.ok) return result.state;

  const { fields } = result;

  try {
    await prisma.recipe.update({
      where: { id, userId: PLACEHOLDER_USER_ID },
      data: fields,
    });
  } catch (err) {
    console.error("updateRecipe failed:", err);
    // P2025 is Prisma's "record not found" code.
    const notFound = (err as { code?: string }).code === "P2025";
    return {
      formError: notFound
        ? "This recipe no longer exists. It may have been deleted."
        : "Couldn't save your changes. Try again.",
      values: { title: fields.title, steps: fields.steps },
    };
  }

  revalidatePath("/recipes");
  revalidatePath(`/recipes/${id}`);
  redirect(`/recipes/${id}`);
}

// Same trust rule as updateRecipe: the id comes from the browser, so
// matching on userId means you can only delete your own recipes.
export async function deleteRecipe(id: string) {
  await requireOwner();

  await prisma.recipe.delete({
    where: { id, userId: PLACEHOLDER_USER_ID },
  });

  revalidatePath("/recipes");
  redirect("/recipes");
}

function toLines(text: string): string[] {
  return text
    .split("\n")
    .map((line) => line.trim())
    .filter(Boolean);
}

// sourceUrl came from user input. Only link to real web pages, never
// something like "javascript:...".
function isWebUrl(value: string | null): value is string {
  return !!value && /^https?:\/\//i.test(value);
}

// An async Server Component: React shows the Suspense fallback while this
// awaits, then streams the finished list into the same HTML response.
async function RecipeListData() {
  const recipes = await prisma.recipe.findMany({
    orderBy: { createdAt: "desc" },
    // Select only what the list needs. Everything passed to a client
    // component is serialized into the page, so smaller is better.
    select: { id: true, title: true, ingredients: true },
  });

  if (recipes.length === 0) {
    return (
      <p className="text-center text-muted">
        No recipes yet.
      </p>
    );
  }
  return <RecipeList recipes={recipes} />;
}

// Only the fields the list needs. The server page selects exactly these,
// so the client never receives the full recipe (steps, image, etc.).
export type RecipeSummary = {
  id: string;
  title: string;
  ingredients: string;
};

type RecipeListProps = {
  recipes: RecipeSummary[];
};

// Write the search to the URL without adding history entries or
// asking the server for anything. An empty search is left out, so an
// unfiltered list stays at plain /recipes.
function syncUrl(search: string) {
  const params = new URLSearchParams(window.location.search);

  if (search.trim()) params.set("q", search);
  else params.delete("q");

  const qs = params.toString();
  window.history.replaceState(
    null,
    "",
    qs ? `?${qs}` : window.location.pathname
  );
}

// Pure helpers, outside the component: easy to read and easy to test.
function matchesSearch(recipe: RecipeSummary, query: string): boolean {
  if (query === "") return true;
  return (
    recipe.title.toLowerCase().includes(query) ||
    recipe.ingredients.toLowerCase().includes(query)
  );
}

// The line under each dish name, like a menu description:
// "ghee, paneer, palak" or "flour, eggs, milk, butter, sugar, and 3 more".
function ingredientPreview(text: string, max = 5): string {
  const items = text
    .split("\n")
    .map((line) => line.trim())
    .filter(Boolean);
  const shown = items.slice(0, max).join(", ");
  const extra = items.length - max;
  return extra > 0 ? `${shown}, and ${extra} more` : shown;
}

export default function RecipeList({ recipes }: RecipeListProps) {
  const searchParams = useSearchParams();

  // Lazy initializer: read the URL once, on the first render.
  // This works on the server too, so a shared link like
  // /recipes?q=egg arrives already filtered, with no flash.
  const [search, setSearch] = useState(() => searchParams.get("q") ?? "");

  // Derived state: calculated on every render, never stored.
  const query = search.trim().toLowerCase();
  const filteredRecipes = recipes.filter((recipe) =>
    matchesSearch(recipe, query)
  );

  const isFiltering = query !== "";

  // The event handler updates state and the URL together. No useEffect:
  // the URL changes because the user did something, so the handler
  // is the right place for it.
  function handleSearchChange(value: string) {
    setSearch(value);
    syncUrl(value);
  }

  function clearSearch() {
    setSearch("");
    syncUrl("");
  }

  return (
    <div>
      <div className="mx-auto max-w-sm">
        <input
          type="search"
          value={search}
          onChange={(e) => handleSearchChange(e.target.value)}
          placeholder="Search by dish or ingredient"
          aria-label="Search recipes"
          className="field"
        />
      </div>

      {/* Always rendered, so screen readers are already watching it when
          the count appears, and the list below doesn't jump. */}
      <p aria-live="polite" className="mt-3 mb-8 h-5 text-center text-sm text-muted">
        {isFiltering && `${filteredRecipes.length} of ${recipes.length} recipes`}
      </p>

      {filteredRecipes.length === 0 ? (
        <div className="text-center">
          <p className="text-muted">No recipes match “{search.trim()}”.</p>
          <button type="button" onClick={clearSearch} className="btn-quiet mt-4">
            Clear search
          </button>
        </div>
      ) : (
        <ul className="menu">
          {filteredRecipes.map((recipe) => {
            const preview = ingredientPreview(recipe.ingredients);
            return (
              <li key={recipe.id}>
                <Link
                  href={`/recipes/${recipe.id}`}
                  className="group block text-center"
                >
                  <h2 className="font-display text-3xl font-medium text-balance transition-colors group-hover:text-basil">
                    {recipe.title}
                  </h2>
                  {preview && (
                    <p className="mt-1.5 text-sm text-muted text-balance">
                      {preview}
                    </p>
                  )}
                </Link>
              </li>

"use client";

import { useReducer } from "react";

type Ingredient = {
  id: string;
  text: string;
};

type Action =
  | { type: "added"; id: string }
  | { type: "changed"; id: string; text: string }
  | { type: "removed"; id: string }
  | { type: "moved"; id: string; direction: "up" | "down" };

type IngredientEditorProps = {
  // Used only to seed the list when the editor first mounts.
  // To load new ingredients later, the parent changes this component's `key`.
  initialIngredients?: string[];
};

// ---------- Reducer ----------
// A pure function: (current state, action) => next state.
// No side effects here: no fetches, no random IDs, no mutation.

function ingredientsReducer(state: Ingredient[], action: Action): Ingredient[] {
  switch (action.type) {
    case "added":
      return [...state, { id: action.id, text: "" }];

    case "changed":
      return state.map((item) =>
        item.id === action.id ? { ...item, text: action.text } : item
      );

    case "removed":
      return state.filter((item) => item.id !== action.id);

    case "moved": {
      const index = state.findIndex((item) => item.id === action.id);
      const target = action.direction === "up" ? index - 1 : index + 1;
      if (index === -1 || target < 0 || target >= state.length) return state;

      const next = [...state];
      [next[index], next[target]] = [next[target], next[index]];
      return next;
    }

    default: {
      // Exhaustiveness check: if you add a new action type and forget
      // to handle it above, this line becomes a type error.
      const unhandled: never = action;
      return unhandled;
    }
  }
}

// Lazy initializer: runs once, on the first render only.
// Fixed IDs (not random ones) keep the server and browser renders identical.
function createInitialState(initialIngredients: string[]): Ingredient[] {
  if (initialIngredients.length === 0) {
    return [{ id: "initial-0", text: "" }];
  }
  return initialIngredients.map((text, i) => ({ id: `initial-${i}`, text }));
}

// ---------- Component ----------

export default function IngredientEditor({
  initialIngredients = [],
}: IngredientEditorProps) {
  const [ingredients, dispatch] = useReducer(
    ingredientsReducer,
    initialIngredients,
    createInitialState
  );

  // Derived state: computed from `ingredients` on every render,
  // never stored separately, so it can never get out of sync.
  const serialized = ingredients
    .map((item) => item.text.trim())
    .filter(Boolean)
    .join("\n");

  return (
    <fieldset>
      <legend className="label">Ingredients</legend>

      {/* This is what the form actually submits, in the same
          newline-separated format your server action already expects. */}
      <input type="hidden" name="ingredients" value={serialized} />

      <ul className="space-y-2">
        {ingredients.map((item, index) => (
          <li key={item.id} className="flex items-center gap-1">
            <input
              type="text"
              value={item.text}
              onChange={(e) =>
                dispatch({ type: "changed", id: item.id, text: e.target.value })
              }
              placeholder="e.g. 2 cups flour"
              aria-label={`Ingredient ${index + 1}`}
              className="field min-w-0 flex-1 mr-1"
            />

            {/* type="button" matters: the default inside a form is
                "submit", which would submit the form on every click. */}
            <button
              type="button"
              onClick={() =>
                dispatch({ type: "moved", id: item.id, direction: "up" })
              }
              disabled={index === 0}
              aria-label={`Move ingredient ${index + 1} up`}
              className="btn-icon"
            >
              ↑
            </button>
            <button
              type="button"
              onClick={() =>
                dispatch({ type: "moved", id: item.id, direction: "down" })
              }
              disabled={index === ingredients.length - 1}
              aria-label={`Move ingredient ${index + 1} down`}
              className="btn-icon"
            >
              ↓
            </button>
            <button
              type="button"
              onClick={() => dispatch({ type: "removed", id: item.id })}
              disabled={ingredients.length === 1}
              aria-label={`Remove ingredient ${index + 1}`}
              className="btn-icon"
            >
              ✕
            </button>
          </li>
        ))}
      </ul>

      <button
        type="button"
        // The ID is created here, in the event handler, not inside the
        // reducer, which keeps the reducer pure.
        onClick={() => dispatch({ type: "added", id: crypto.randomUUID() })}
        className="btn-quiet mt-3"
      >
        Add ingredient
      </button>
    </fieldset>
  );
}

      </div>

      {/* The editor keeps its own state, so the reset doesn't touch it. */}
      <div>
        <IngredientEditor
          initialIngredients={splitIngredients(initialValues.ingredients)}
        />
        {errors.ingredients && (
          <p role="alert" className="field-error">
            {errors.ingredients}
          </p>
        )}
      </div>

      <div>
        <label htmlFor="steps" className="label">
          Steps
        </label>
        <textarea
          id="steps"
          name="steps"
          rows={8}
          defaultValue={state.values?.steps ?? initialValues.steps}
          placeholder="One step per line"
          aria-invalid={errors.steps ? true : undefined}
          aria-describedby={errors.steps ? "steps-error" : undefined}
          className="field leading-relaxed"
        />
        {errors.steps && (
          <p id="steps-error" className="field-error">
            {errors.steps}
          </p>
        )}
      </div>

      <div className="border-t border-rule pt-6">
        <button type="submit" disabled={isPending} className="btn-primary">
          {isPending ? pendingLabel : submitLabel}
        </button>
      </div>
    </form>
  );
}

import * as cheerio from "cheerio";

interface ScrapedRecipe {
  title: string;
  imageUrl: string | null;
  ingredients: string;
  steps: string;
}

export async function scrapeRecipeFromUrl(url: string): Promise<ScrapedRecipe> {
  const res = await fetch(url, {
    headers: { "User-Agent": "Mozilla/5.0 (compatible; BurpTalesBot/1.0)" },
  });
  if (!res.ok) throw new Error(`Failed to fetch page (status ${res.status})`);

  const html = await res.text();
  const $ = cheerio.load(html);

  // Most recipe sites embed structured "Recipe" data as JSON-LD for Google.
  // We look for that first since it's far more reliable than guessing from HTML.
  let recipeData: any = null;
  $('script[type="application/ld+json"]').each((_, el) => {
    if (recipeData) return;
    try {
      const json = JSON.parse($(el).text());
      const candidates = Array.isArray(json)
        ? json
        : json["@graph"]
        ? json["@graph"]
        : [json];
      const found = candidates.find((item: any) => {
        const type = item["@type"];
        return type === "Recipe" || (Array.isArray(type) && type.includes("Recipe"));
      });
      if (found) recipeData = found;
    } catch {
      // Some sites ship malformed JSON-LD — just skip it
    }
  });

  if (recipeData) {
    const ingredients: string[] = recipeData.recipeIngredient || [];
    const rawInstructions = recipeData.recipeInstructions || [];
    const instructions: string[] = Array.isArray(rawInstructions)
      ? rawInstructions.map((step: any) =>
          typeof step === "string" ? step : step.text || ""
        )
      : [String(rawInstructions)];

    return {
      title: recipeData.name || $("title").text() || "Untitled recipe",
      imageUrl: Array.isArray(recipeData.image)
        ? recipeData.image[0]
        : recipeData.image?.url || recipeData.image || null,
      ingredients: ingredients.join("\n"),
      steps: instructions.join("\n"),
    };
  }

  // Fallback if the site has no structured recipe data at all
  return {
    title: $("title").text() || "Untitled recipe",
    imageUrl: $('meta[property="og:image"]').attr("content") || null,
    ingredients: "",
    steps: "",
  };
}

===== prisma/seed.ts
/**
 * Seeds the database with demo recipes and restaurant visits for the
 * showcase site.
 *
 * Run with:  npm run seed
 *
 * Safe to run more than once: it first deletes the demo rows it created
 * last time (matched by recipe title and restaurant name), then inserts
 * them again. Anything you added yourself is left alone.
 */
import { PrismaClient } from "@prisma/client";
import { recipes } from "./seed-data/recipes";
import { restaurants } from "./seed-data/restaurants";

const prisma = new PrismaClient();

const DAY_MS = 24 * 60 * 60 * 1000;

function daysAgo(days: number): Date {
  return new Date(Date.now() - days * DAY_MS);
}

// Recipes and visits need a user. The app still uses a placeholder user,
// and edits only match recipes owned by it, so the seed must use the
// same one. Set SEED_USER_ID to that id if you have more than one user.
async function getUserId(): Promise<string> {
  const fromEnv = process.env.SEED_USER_ID;
  if (fromEnv) {
    const user = await prisma.user.findUnique({ where: { id: fromEnv } });
    if (!user) throw new Error(`SEED_USER_ID ${fromEnv} doesn't match any user`);
    return user.id;
  }

  const firstUser = await prisma.user.findFirst({ orderBy: { createdAt: "asc" } });
  if (firstUser) return firstUser.id;

  const created = await prisma.user.create({
    data: { email: "demo@burptales.dev", name: "Demo Cook" },
  });
  return created.id;
}

async function clearDemoData(userId: string) {
  const recipeTitles = recipes.map((r) => r.title);
  const restaurantNames = restaurants.map((r) => r.name);

  // Recipe-tag links are removed automatically with the recipe.
  const deletedRecipes = await prisma.recipe.deleteMany({
    where: { userId, title: { in: recipeTitles } },
  });

  // Visits point at restaurants, so they go first.
  await prisma.visit.deleteMany({
    where: { restaurant: { name: { in: restaurantNames } } },
  });
  const deletedRestaurants = await prisma.restaurant.deleteMany({
    where: { name: { in: restaurantNames } },
  });

  console.log(
    `Cleared ${deletedRecipes.count} old demo recipes and ${deletedRestaurants.count} old demo restaurants.`
  );
}

async function seedRecipes(userId: string) {
  // One at a time on purpose: two recipes creating the same new tag in
  // parallel would collide on the tag's unique (name, type) index.
  for (const [index, recipe] of recipes.entries()) {
    await prisma.recipe.create({
      data: {
        userId,
        title: recipe.title,
        ingredients: recipe.ingredients.join("\n"),
        steps: recipe.steps.join("\n"),
        cookedCount: recipe.cookedCount ?? 0,
        lastCookedAt:
          recipe.lastCookedDaysAgo !== undefined
            ? daysAgo(recipe.lastCookedDaysAgo)
            : null,
        // Spread out over the last two months, so "recent recipes"
        // and the default newest-first sort look natural.
        createdAt: daysAgo(index * 2),
        tags: {
          connectOrCreate: recipe.tags.map((name) => ({
            where: { name_type: { name, type: "recipe" } },
            create: { name, type: "recipe" },
          })),
        },
      },
    });
  }
  console.log(`Created ${recipes.length} recipes.`);
}

async function seedRestaurants(userId: string) {
  let visitCount = 0;
  for (const restaurant of restaurants) {
    await prisma.restaurant.create({
      data: {
        name: restaurant.name,
        cuisine: restaurant.cuisine,
        city: restaurant.city,
        visits: {
          create: restaurant.visits.map((visit) => ({
            userId,
            date: new Date(`${visit.date}T19:00:00Z`),
            rating: visit.rating,
            notes: visit.notes,
            dishesTried: visit.dishesTried.join("\n"),
          })),
        },
      },
    });
    visitCount += restaurant.visits.length;
  }
  console.log(`Created ${restaurants.length} restaurants with ${visitCount} visits.`);
}

async function main() {
  const userId = await getUserId();
  console.log(`Seeding demo data for user ${userId}`);
  await clearDemoData(userId);
  await seedRecipes(userId);
  await seedRestaurants(userId);
  console.log("Done.");
}

main()
  .catch((error) => {
    console.error(error);
    process.exitCode = 1;
  })
  .finally(() => prisma.$disconnect());
